# Structure of project

```
backend/ 
├── cmd/
│ └── app/
│ └── main.go # Точка входа. Инициализация и запуск сервера
├── config/
│ └── config.go # Загрузка env-переменных 
├── docs/ # API
├── internal/ # Приватный код приложения (нельзя импортировать извне)
│ ├── domain/ # Чистые сущности бизнеса, интерфейсы (без внешних библиотек)
│ │  ├── user.go # Структура пользователя и распределение ролей
│ │  ├── order.go # Структура заказов, отслеживание статуса
│ │  ├── chat.go # Структура сообщений и чата
│ │  └── analytics.go # Структуры метрик и KPI
│ │
│ ├── usecase/ # Бизнес-логика (Оркестрация сценариев)
│ │ ├── auth/ # Сценарий авторизации через тг + генерация JWT
│ │ ├── order/ # Сценарий создания заказа, смена статусов, логирование
│ │ ├── chat/ # Сценарий отправки сообщения, логика перехвата чата
│ │ └── analytics/ # Сценарий расчета KPI сотрудников и графиков продаж
│ │
│ ├── repository/ # Инфраструктура данных (Реализация интерфейсов)
│ │ └── postgres/ # Работа с PostgreSQL
│ │ ├── user_pg.go # Логика пользователя и связь с БД
│ │ ├── order_pg.go # Логика заказа и связь с БД
│ │ └── chat_pg.go # Логика чата и связь с БД
│ │
│ └── delivery/ # Транспортный уровень (Входные точки)
│ ├── http/ # REST API
│ │ ├── v1/
│ │ │ ├── router.go
│ │ │ ├── auth.go # Хэндлеры для авторизации
│ │ │ └── order.go # Хэндлеры для заказов и аналитики
│ │ └── middleware/ # Нарезка прав для сотрудников
│ └── ws/ # WebSocket сервер (управление real-time пулом соединений)
│ ├── hub.go # Менеджер активных коннектов сотрудников и клиентов
│ └── handler.go # Чтение/запись фреймов в сокеты
│
├── migrations/ # SQL-файлы миграций
├── go.mod
└── go.sum

frontend/ 
├── public/ 
├── src/
│ ├── app/ # Инициализация приложения
│ │ ├── providers/ # React-контексты (ThemeContext от Telegram, React Query, AuthProvider)
│ │ ├── routers/ # Настройка роутинга (доступ к экранам по ролям)
│ │ ├── styles/ # Глобальные стили и Tailwind-конфиги
│ │ └── index.tsx # Точка монтирования React
│ │
│ ├── pages/ # Композиция экранов из фич и виджетов
│ │ ├── client-shop/ # Экран магазина для клиента
│ │ ├── employee-desk/ # Рабочее место сотрудника (чаты + его KPI)
│ │ └── manager-dashboard/ # Панель руководителя (аналитика всей команды)
│ │
│ ├── widgets/ # Крупные самостоятельные блоки страниц
│ │ ├── chat-window/ # Виджет чата (история + инпут)
│ │ ├── kpi-tracker/ # Виджет с таблицами и метриками эффективности
│ │ └── analytics-charts/ # Блок с графиками продаж (Recharts)
│ │
│ ├── features/ # Действия пользователя, несущие бизнес-ценность
│ │ ├── order-checkout/ # Кнопка и логика оформления заказа клиентом
│ │ ├── chat-takeover/ # Кнопка «Перехватить диалог» для сотрудника
│ │ └── change-order-status/# Выпадающий список смены статуса CRM-сделки
│ │
│ ├── entities/ # Бизнес-сущности (только отображение данных и их стейт)
│ │ ├── user/ # Модели User, хуки получения профиля, стейт текущей роли
│ │ ├── order/ # Карточка заказа, логика расчета стоимости корзины
│ │ └── chat/ # Элемент сообщения, структура диалога
│ │
│ └── shared/ # Переиспользуемый инфраструктурный код (UI-кит)
│ ├── api/ # Клиент для запросов (Axios / Fetch + WebSocket клиент)
│ ├── lib/ # Telegram WebApp SDK хелперы, форматирование дат/валют
│ └── ui/ # Глупые компоненты (Button, Input, Loader, Badge)
```
